## 2. PostgreSQL DDL (Data Definition Language)

> The source of truth is the Alembic migration `backend/alembic/versions/0001_initial_schema.py`
> (models in `backend/vibeguard/db/models.py`). This DDL mirrors it. UUIDs are generated by the
> application, so no `uuid-ossp` extension is needed.

```sql
-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Severity Levels Enum
CREATE TYPE vulnerability_severity AS ENUM (
    'CRITICAL',
    'HIGH',
    'MEDIUM',
    'LOW',
    'INFO'
);

-- Scan Job Status Enum
CREATE TYPE scan_status AS ENUM (
    'QUEUED',
    'CLONING',
    'ANALYZING_STATIC',
    'ANALYZING_LLM',
    'COMPLETED',
    'FAILED'
);

-- Finding Remediation Status Enum
CREATE TYPE finding_status AS ENUM (
    'OPEN',
    'PATCH_APPLIED',
    'DISMISSED_FALSE_POSITIVE',
    'RESOLVED'
);

-- Scan Source Enum
CREATE TYPE scan_source AS ENUM (
    'GITHUB',
    'ZIP',
    'LOCAL'          -- server-side path; disabled unless VIBEGUARD_ALLOW_LOCAL_PATHS + scan root
);

-- Finding Origin Enum
CREATE TYPE finding_source AS ENUM ('STATIC', 'LLM', 'MERGED');

-- Repositories Table
CREATE TABLE repositories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    url VARCHAR(512) NOT NULL UNIQUE,  -- canonical, lowercased GitHub URL
    default_branch VARCHAR(64) DEFAULT 'main',
    is_private BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Scans Table
CREATE TABLE scans (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    repo_id UUID NOT NULL REFERENCES repositories(id) ON DELETE CASCADE,
    source_type scan_source NOT NULL DEFAULT 'GITHUB',
    commit_hash VARCHAR(40),          -- NULL for ZIP uploads
    status scan_status DEFAULT 'QUEUED',
    hardening_score INTEGER CHECK (hardening_score >= 0 AND hardening_score <= 100),
    total_findings INTEGER DEFAULT 0,
    critical_count INTEGER DEFAULT 0,
    high_count INTEGER DEFAULT 0,
    medium_count INTEGER DEFAULT 0,
    low_count INTEGER DEFAULT 0,      -- score + counts cover OPEN findings only (triage moves them)
    files_scanned INTEGER,
    llm_used BOOLEAN DEFAULT FALSE,
    warnings JSONB DEFAULT '[]',      -- e.g. "LLM pass disabled", "chunk skipped: deadline"
    duration_ms INTEGER,
    error_message TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP WITH TIME ZONE
);

-- Findings Table
CREATE TABLE findings (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    scan_id UUID NOT NULL REFERENCES scans(id) ON DELETE CASCADE,
    rule_id VARCHAR(128) NOT NULL,    -- e.g. 'stripe-secret-key', 'llm-broken-access-control'
    severity vulnerability_severity NOT NULL,
    category VARCHAR(128) NOT NULL, -- e.g., 'Broken Authentication', 'Secret Exposure'
    title VARCHAR(255) NOT NULL,
    cwe_id VARCHAR(32),              -- e.g., 'CWE-639', 'CWE-798'
    file_path VARCHAR(512) NOT NULL,
    line_start INTEGER NOT NULL,
    line_end INTEGER NOT NULL,
    snippet_original TEXT NOT NULL,  -- ±3 lines only; secrets redacted
    snippet_fixed TEXT,              -- NULL when no code-level fix exists (e.g. rotate a leaked key)
    unified_diff TEXT,               -- NULL when snippet_fixed is NULL; masked (display-only) for secrets
    explanation TEXT NOT NULL,
    source finding_source NOT NULL,
    status finding_status DEFAULT 'OPEN',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Remediation coordinates (1:1, optional). Lets the Phase 3 PR worker patch a fresh clone for
-- secret findings, whose stored diff is masked. No secret value is stored: before replacing,
-- the worker checks line[col_start:col_end] equals expected_text, or re-matches verify_rule.
CREATE TABLE remediations (
    id UUID PRIMARY KEY,
    finding_id UUID NOT NULL UNIQUE REFERENCES findings(id) ON DELETE CASCADE,
    kind VARCHAR(32) NOT NULL,        -- 'replace_span'
    file_path VARCHAR(512) NOT NULL,
    line INTEGER NOT NULL,            -- 1-based
    col_start INTEGER NOT NULL,       -- 0-based, inclusive
    col_end INTEGER NOT NULL,         -- 0-based, exclusive
    replacement TEXT NOT NULL,        -- e.g. 'process.env.STRIPE_SECRET_KEY!'
    env_var_name VARCHAR(128),        -- e.g. 'STRIPE_SECRET_KEY'
    expected_text TEXT,               -- for non-secret spans (env var names)
    verify_rule VARCHAR(128)          -- for secret spans: rule id to re-match
);

-- Pull Requests opened from finding fixes (migration 0002; one per finding)
CREATE TABLE pull_requests (
    id UUID PRIMARY KEY,
    finding_id UUID NOT NULL UNIQUE REFERENCES findings(id) ON DELETE CASCADE,
    pr_number INTEGER NOT NULL,
    pr_url VARCHAR(512) NOT NULL,
    branch_name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Indexes for Fast Querying
CREATE INDEX idx_scans_repo_id ON scans(repo_id);
CREATE INDEX idx_scans_status ON scans(status);
CREATE INDEX idx_findings_scan_id ON findings(scan_id);
CREATE INDEX idx_findings_severity ON findings(severity);
```

## 3. Redis Key Structure (Caching & Ephemeral Jobs)

| Key Pattern | Data Type | TTL | Purpose |
| :--- | :--- | :--- | :--- |
| `scan:job:{scan_id}` | Hash | 2 hours | Live progress state (`step`, `percentage`, `status`, `worker_id`, `updated_at`). |
| `scan:events:{scan_id}` | Pub/Sub channel | — | JSON events: `PROGRESS`, `SCAN_COMPLETED`, `SCAN_FAILED` (WebSocket fan-out in Phase 3). |
| `scan:lock:{repo_url}` | String | 5 minutes | Concurrency lock (value = owning `scan_id`); the API takes it, the worker releases it. Key uses the canonical URL. |
| `npm:package:{pkg_name}` / `pypi:package:{pkg_name}` | String | 24 hours | `"1"`/`"0"` registry existence cache (reduces registry calls). |
